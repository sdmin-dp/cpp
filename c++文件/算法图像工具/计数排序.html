void counting_sort(int *arr) {
    int maxValue = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > maxValue) maxValue = arr[i];
    }

    int count[maxValue + 1] = {0};
    for (int i = 0; i < n; i++) {
        count[arr[i]]++;
    }

    int k = 0;
    for (int value = 0; value <= maxValue; value++) {
        while (count[value] > 0) {
            arr[k] = value;
            count[value]--;
            k++;
        }
    }
}
