// LSD 十进制基数排序：每一位用稳定计数排序
void radix_sort(int arr[], int n) {
    int maxValue = arr[0];
    for (int i = 1; i < n; i++)
        if (arr[i] > maxValue) maxValue = arr[i];

    for (int exp = 1; maxValue / exp > 0; exp *= 10) {
        int output[105];
        int count[10] = {0};

        for (int i = 0; i < n; i++)
            count[(arr[i] / exp) % 10]++;

        for (int i = 1; i < 10; i++)
            count[i] += count[i - 1];

        for (int i = n - 1; i >= 0; i--) {
            int digit = (arr[i] / exp) % 10;
            output[count[digit] - 1] = arr[i];
            count[digit]--;
        }

        for (int i = 0; i < n; i++)
            arr[i] = output[i];
    }
}
