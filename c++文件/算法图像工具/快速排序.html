// 快速排序：本页按这段代码演示
void kuaipai(int a[], int l, int r) {
    if (l < r) {
        int m = a[l], i = l, j = r;
        while (i <= j) {
            while (a[i] < m) i++;
            while (a[j] > m) j--;
            if (i <= j) swap(a[i++], a[j--]);
        }
        if (l < j) kuaipai(a, l, j);
        if (r > i) kuaipai(a, i, r);
    }
}
