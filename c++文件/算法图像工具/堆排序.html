void Down(int a[], int n, int parent) {
    int child = parent * 2 + 1;
    while (child < n) {
        if (child + 1 < n && a[child + 1] > a[child])
            child++;
        if (a[child] > a[parent]) {
            swap(a[child], a[parent]);
            parent = child;
            child = parent * 2 + 1;
        }
        else
            return;
    }
}

void HeapSort(int a[], int n) {
    for (int i = (n - 2) / 2; i >= 0; i--)
        Down(a, n, i);
    int end = n - 1;
    while (end > 0) {
        swap(a[0], a[end]);
        Down(a, end, 0);
        end--;
    }
}
