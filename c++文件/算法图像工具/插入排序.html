void insertion_sort(int * arr) {
    for (int i = 1; i < n; i++) {
        int key = arr[i];    // 存储待排序元素
        int j = i - 1;
        while ((j >= 0) && (key < arr[j])) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}
