// 归并排序：本页按这段代码的切分与合并来演示
void merge(int * a, int l, int r) {
    if (r > l) {
        int b[105];
        int m = l + ((r - l) >> 1);
        merge(a, l, m);
        merge(a, m + 1, r);

        int i = l, j = m + 1;
        int t = l;
        while (i <= m && j <= r) {
            if (a[i] <= a[j]) b[t++] = a[i++];
            else b[t++] = a[j++];
        }
        while (i <= m) b[t++] = a[i++];
        while (j <= r) b[t++] = a[j++];
        for (int i = l; i <= r; i++) a[i] = b[i];
    }
}
